    <!-- FIREBASE AUTHENTICATION SCRIPT -->
    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { 
            getAuth, 
            signInAnonymously, 
            signInWithCustomToken, 
            GoogleAuthProvider, 
            signInWithPopup, 
            onAuthStateChanged, 
            signOut,
            updateProfile 
        } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";

        // Konfigurasi Firebase dari Environment atau Fallback Demo
        const firebaseConfig = typeof __firebase_config !== 'undefined' 
            ? JSON.parse(__firebase_config) 
            : {
                apiKey: "AIzaSyDemoConfigKeyForFotoCloud2026",
                authDomain: "fotocloud-demo.firebaseapp.com",
                projectId: "fotocloud-demo",
                storageBucket: "fotocloud-demo.appspot.com",
                messagingSenderId: "123456789",
                appId: "1:123456789:web:fotocloud2026"
            };

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);

        // Auto authenticate dengan token jika ada dari lingkungan server
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
            signInWithCustomToken(auth, __initial_auth_token).catch((err) => {
                console.warn("Custom token auth fallback to anonymous:", err);
            });
        }

        // Listen Perubahan Status Autentikasi Firebase
        onAuthStateChanged(auth, (user) => {
            if (user) {
                const displayName = user.displayName || user.email?.split('@')[0] || localStorage.getItem('user_name') || 'Pengguna';
                localStorage.setItem('user_id', user.uid);
                localStorage.setItem('user_name', displayName);
                updateUI(user.uid, displayName);
            } else {
                // Jika logout dari Firebase Auth, periksa fallback localStorage
                const localId = localStorage.getItem('user_id');
                const localName = localStorage.getItem('user_name');
                if (localId && localName) {
                    updateUI(localId, localName);
                } else {
                    updateUI(null, '');
                }
            }
        });

        // Handler Login Google Firebase
        window.loginWithGoogle = async function() {
            const provider = new GoogleAuthProvider();
            try {
                const result = await signInWithPopup(auth, provider);
                const user = result.user;
                const name = user.displayName || user.email?.split('@')[0] || 'User Google';
                localStorage.setItem('user_id', user.uid);
                localStorage.setItem('user_name', name);
                closeAuthModal();
                updateUI(user.uid, name);
            } catch (error) {
                console.warn("Google popup error, fallback to demo mode:", error);
                // Fallback login jika popup google diblokir iFrame
                window.loginDemo('google_demo', 'Google User');
            }
        };

        // Handler Login Demo / Anonymous via Firebase
        window.loginDemo = async function(id, name) {
            try {
                if (!auth.currentUser) {
                    await signInAnonymously(auth);
                }
                if (auth.currentUser) {
                    await updateProfile(auth.currentUser, { displayName: name });
                    localStorage.setItem('user_id', auth.currentUser.uid);
                    localStorage.setItem('user_name', name);
                    updateUI(auth.currentUser.uid, name);
                }
            } catch (err) {
                console.warn("Firebase Anonymous Auth fallback:", err);
                localStorage.setItem('user_id', id);
                localStorage.setItem('user_name', name);
                updateUI(id, name);
            }
            closeAuthModal();
        };

        // Handler Form Custom Username Login
        window.handleCustomLogin = async function(e) {
            e.preventDefault();
            const inputName = document.getElementById('custom-username').value.trim();
            if (inputName) {
                const customId = 'user_' + inputName.toLowerCase().replace(/\s+/g, '_');
                await window.loginDemo(customId, inputName);
            }
        };

        // Logout Sesi Firebase & LocalStorage
        window.logout = async function() {
            try {
                await signOut(auth);
            } catch (e) {
                console.warn("SignOut error:", e);
            }
            localStorage.removeItem('user_id');
            localStorage.removeItem('user_name');
            updateUI(null, '');
        };

        // Simpan Nama Baru
        window.saveNewName = async function(e) {
            e.preventDefault();
            const newName = document.getElementById('new-name-input').value.trim();
            if (newName) {
                if (auth.currentUser) {
                    try {
                        await updateProfile(auth.currentUser, { displayName: newName });
                    } catch (err) {
                        console.warn("Update profile error:", err);
                    }
                }
                localStorage.setItem('user_name', newName);
                closeRenameModal();
                updateUI(localStorage.getItem('user_id'), newName);
            }
        };

        function updateUI(userId, userName) {
            const navContainer = document.getElementById('nav-right-container');
            const banner = document.getElementById('logged-in-banner');
            const bannerUserName = document.getElementById('banner-user-name');
            const heroCtaBtn = document.getElementById('hero-cta-btn');

            if (userId) {
                if (banner) {
                    banner.classList.remove('hidden');
                    if (bannerUserName) bannerUserName.textContent = userName;
                }

                if (navContainer) {
                    navContainer.innerHTML = `
                        <div class="hidden sm:flex flex-col items-end mr-1 text-right">
                            <span class="text-[8px] uppercase tracking-widest text-gray-400 font-bold">Terpakai</span>
                            <div class="flex items-center gap-1 text-xs font-bold text-[var(--gold)]">
                                <i data-lucide="hard-drive" class="w-3 h-3"></i>
                                <span>3GB / 15GB</span>
                            </div>

        // Global UI Event Helpers
        window.toggleFaq = function(id) {
            const content = document.getElementById(`faq-content-${id}`);
            const icon = document.getElementById(`faq-icon-${id}`);
            if (content) {
                const isHidden = content.classList.contains('hidden');
                document.querySelectorAll('[id^="faq-content-"]').forEach(el => el.classList.add('hidden'));
                document.querySelectorAll('[id^="faq-icon-"]').forEach(el => el.style.transform = 'rotate(0deg)');

                if (isHidden) {
                    content.classList.remove('hidden');
                    if (icon) icon.style.transform = 'rotate(180deg)';
                }
            }
        };

        window.toggleDropdown = function() {
            const dropdown = document.getElementById('user-dropdown');
            if (dropdown) dropdown.classList.toggle('hidden');
        };

        window.triggerChangeName = function(currentName) {
            const dropdown = document.getElementById('user-dropdown');
            if (dropdown) dropdown.classList.add('hidden');
            openRenameModal(currentName);
        };

        window.openRenameModal = function(currentName) {
            const modal = document.getElementById('rename-modal');
            const input = document.getElementById('new-name-input');
            input.value = currentName || localStorage.getItem('user_name') || '';
            modal.classList.remove('hidden');
        };

        window.closeRenameModal = function() {
            document.getElementById('rename-modal').classList.add('hidden');
        };

        window.openAuthModal = function(mode) {
            const modal = document.getElementById('auth-modal');
            const modalTitle = document.getElementById('modal-title');
            if (modalTitle) {
                modalTitle.textContent = mode === 'register' ? 'Daftar Akun FotoCloud' : 'Masuk ke FotoCloud';
            }
            modal.classList.remove('hidden');
        };

        window.closeAuthModal = function() {
            document.getElementById('auth-modal').classList.add('hidden');
        };

        window.scrollToDemo = function() {
            const demoSec = document.getElementById('demo-section');
            if (demoSec) demoSec.scrollIntoView({ behavior: 'smooth' });
        };

        window.openUpgradeModal = function() {
            document.getElementById('upgrade-modal').classList.remove('hidden');
        };

        window.closeUpgradeModal = function() {
            document.getElementById('upgrade-modal').classList.add('hidden');
        };

        function escapeHtml(text) {
            return String(text)
                .replace(/&/g, "&amp;")
                .replace(/</g, "&lt;")
                .replace(/>/g, "&gt;")
                .replace(/"/g, "&quot;")
                .replace(/'/g, "&#039;");
        }

        // Close dropdown when clicking outside
        window.addEventListener('click', (e) => {
            const navRight = document.getElementById('nav-right-container');
            if (navRight && !navRight.contains(e.target)) {
                const dropdown = document.getElementById('user-dropdown');
                if (dropdown) dropdown.classList.add('hidden');
            }
        });

        // Initialize Lucide icons on DOM ready
        document.addEventListener('DOMContentLoaded', () => {
            if (window.lucide) lucide.createIcons();
        });
